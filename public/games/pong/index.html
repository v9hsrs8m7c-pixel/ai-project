How to Play

Controls: Use W/S keys (Player 1) or Up/Down arrows (Player 2)

Important: Click on the game area first to enable controls!

Multiplayer: One player hosts a game, others connect using the server URL

Objective: First to 10 points wins!

Server Setup

To host a game, you need to run the WebSocket server. Copy this code into a file called server.js and run node server.js:

const WebSocket = require('ws'); const port = 8080; // Port for the WebSocket server const host = "0.0.0.0"; // Host for the WebSocket server const server = new WebSocket.Server({ port: port, host: host }); let gameState = { ball: { x: 400, y: 200, dx: 5, dy: 3, radius: 8 }, paddle1: { x: 20, y: 150, width: 10, height: 100 }, paddle2: { x: 770, y: 150, width: 10, height: 100 }, score1: 0, score2: 0, gameStarted: false, ballSpeed: 5, // Track current ball speed speedIncrement: 0.3 // How much to increase speed each hit }; let players = []; let gameInterval; function broadcast(data) { players.forEach(player => { if (player.ws.readyState === WebSocket.OPEN) { player.ws.send(JSON.stringify(data)); } }); } function updateGame() { if (!gameState.gameStarted || players.length < 2) return; // Update ball position gameState.ball.x += gameState.ball.dx; gameState.ball.y += gameState.ball.dy; // Ball collision with top/bottom walls if (gameState.ball.y <= gameState.ball.radius || gameState.ball.y >= 400 - gameState.ball.radius) { gameState.ball.dy = -gameState.ball.dy; } // Ball collision with paddles const ball = gameState.ball; const p1 = gameState.paddle1; const p2 = gameState.paddle2; // Left paddle collision if (ball.x - ball.radius <= p1.x + p1.width && ball.y >= p1.y && ball.y <= p1.y + p1.height && ball.dx < 0) { ball.dx = -ball.dx; ball.x = p1.x + p1.width + ball.radius; // Increase ball speed on paddle hit gameState.ballSpeed += gameState.speedIncrement; const speedMultiplier = gameState.ballSpeed / 5; // 5 is the initial speed ball.dx = ball.dx > 0 ? gameState.ballSpeed : -gameState.ballSpeed; ball.dy = ball.dy * speedMultiplier; } // Right paddle collision if (ball.x + ball.radius >= p2.x && ball.y >= p2.y && ball.y <= p2.y + p2.height && ball.dx > 0) { ball.dx = -ball.dx; ball.x = p2.x - ball.radius; // Increase ball speed on paddle hit gameState.ballSpeed += gameState.speedIncrement; const speedMultiplier = gameState.ballSpeed / 5; // 5 is the initial speed ball.dx = ball.dx > 0 ? gameState.ballSpeed : -gameState.ballSpeed; ball.dy = ball.dy * speedMultiplier; } // Scoring if (ball.x < 0) { gameState.score2++; resetBall(); } else if (ball.x > 800) { gameState.score1++; resetBall(); } // Check for game end if (gameState.score1 >= 10 || gameState.score2 >= 10) { gameState.gameStarted = false; broadcast({ type: 'gameEnd', winner: gameState.score1 >= 10 ? 1 : 2 }); } broadcast({ type: 'gameUpdate', gameState }); } function resetBall() { gameState.ball.x = 400; gameState.ball.y = 200; // Reset ball speed to initial value on scoring gameState.ballSpeed = 5; gameState.ball.dx = Math.random() > 0.5 ? 5 : -5; gameState.ball.dy = (Math.random() - 0.5) * 6; } function resetGame() { gameState.score1 = 0; gameState.score2 = 0; gameState.ballSpeed = 5; resetBall(); gameState.gameStarted = players.length === 2; broadcast({ type: 'gameReset' }); } server.on('connection', (ws) => { console.log('New player connected'); const playerId = players.length < 2 ? players.length + 1 : null; if (playerId) { players.push({ ws, id: playerId }); ws.send(JSON.stringify({ type: 'playerAssigned', playerId })); if (players.length === 2) { gameState.gameStarted = true; broadcast({ type: 'gameStart' }); gameInterval = setInterval(updateGame, 1000/60); } } else { ws.send(JSON.stringify({ type: 'gameFull' })); ws.close(); } ws.on('message', (message) => { try { const data = JSON.parse(message); if (data.type === 'paddleUpdate') { const player = players.find(p => p.ws === ws); if (player) { if (player.id === 1) { gameState.paddle1.y = Math.max(0, Math.min(300, data.y)); } else if (player.id === 2) { gameState.paddle2.y = Math.max(0, Math.min(300, data.y)); } } } else if (data.type === 'newGame') { // Allow any connected player to reset the game resetGame(); } } catch (e) { console.error('Invalid message:', e); } }); ws.on('close', () => { console.log('Player disconnected'); players = players.filter(p => p.ws !== ws); if (players.length < 2) { gameState.gameStarted = false; if (gameInterval) { clearInterval(gameInterval); gameInterval = null; } } if (players.length === 0) { gameState.score1 = 0; gameState.score2 = 0; resetBall(); } }); }); console.log(`Pong WebSocket server is running on ws://${host}:${port}`);
